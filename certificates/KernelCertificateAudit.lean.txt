import Lean
import «NSTorusUniqueness»
import «NSTorusUniqueness».«CellIntegral»
import «NSTorusUniqueness».«CellLebesgue»
import «NSTorusUniqueness».«Clamp»
import «NSTorusUniqueness».«Defs»
import «NSTorusUniqueness».«Energy»
import «NSTorusUniqueness».«FieldIdentities»
import «NSTorusUniqueness».«IteratedIntegral»
import «NSTorusUniqueness».«Main»
import «NSTorusUniqueness».«Uniqueness»

open Lean Elab Command

/-- Prints `#print axioms`-style lines for every theorem declared in the listed modules. -/
elab "#kernel_certificate_audit" : command => do
  let env ← getEnv
  let mods : Array Name := #[`«NSTorusUniqueness», `«NSTorusUniqueness».«CellIntegral», `«NSTorusUniqueness».«CellLebesgue», `«NSTorusUniqueness».«Clamp», `«NSTorusUniqueness».«Defs», `«NSTorusUniqueness».«Energy», `«NSTorusUniqueness».«FieldIdentities», `«NSTorusUniqueness».«IteratedIntegral», `«NSTorusUniqueness».«Main», `«NSTorusUniqueness».«Uniqueness»]
  for m in mods do
    match env.getModuleIdx? m with
    | none => IO.println s!"MODULE NOT LOADED {m}"
    | some i =>
      for n in (env.header.moduleData[i.toNat]!).constNames do
        match env.find? n with
        | some (.thmInfo _) =>
          if !n.isInternal then
            let ax ← collectAxioms n
            let axs := ", ".intercalate (ax.toList.map toString)
            if ax.isEmpty then IO.println s!"'{n}' does not depend on any axioms [module {m}]"
            else IO.println s!"'{n}' depends on axioms: [{axs}] [module {m}]"
        | _ => pure ()

#kernel_certificate_audit
